SELECT PRDCLC.* 
,CLC.nombre as ciclo_nombre
,PRV.nombre as proveedor_nombre
,PRD.nombre as producto_nombre
FROM public."producto_ciclo"  PRDCLC

INNER JOIN public.ciclos_compra CLC
ON PRDCLC.ciclo_id = CLC.id

INNER JOIN public.proveedores PRV
ON PRDCLC.proveedor_id = PRV.id

INNER JOIN public.productos PRD
ON PRDCLC.producto_id = PRD.id
