SELECT 
    pc.id AS pedido_id,
    p.nombre AS proveedor,
    pc.fecha_pedido,
    pc.estado,
    pc.observaciones
FROM pedidos_proveedores_cabecera pc
JOIN proveedores p ON pc.proveedor_id = p.id
WHERE pc.ciclo_id = {{appsmith.store.ciclo_actual_id}} -- Asumiendo que guardas el ciclo activo
ORDER BY p.nombre;
