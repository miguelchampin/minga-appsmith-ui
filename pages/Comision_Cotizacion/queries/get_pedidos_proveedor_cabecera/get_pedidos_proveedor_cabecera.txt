SELECT 
    pc.id AS pedido_id,
    p.nombre AS proveedor,
    pc.fecha_pedido,
    pc.estado,
    pc.observaciones
FROM pedidos_proveedores_cabecera pc

JOIN proveedores p 
ON pc.proveedor_id = p.id

JOIN(
	SELECT *
	FROM ciclos_compra
	WHERE fecha_apertura in (select max(fecha_apertura) from ciclos_compra)
)c
on pc.ciclo_id = c.id

ORDER BY p.nombre;
