SELECT 
    pd.id AS detalle_id,
    prod.nombre AS producto,
    pd.cantidad_solicitada,
    pd.cantidad_recibida
		
FROM pedidos_proveedores_detalle pd

JOIN productos_ciclo pc 
ON pd.productos_ciclo_id = pc.id

JOIN productos prod 
ON pc.producto_id = prod.id

WHERE pd.pedidos_proveedores_cabecera_id = {{Tabla_proveedor_pedidos.selectedRow.pedido_id}};
