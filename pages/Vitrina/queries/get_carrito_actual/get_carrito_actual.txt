SELECT 
  ped.id as detalle_id 
  ,p.nombre
  ,ped.cantidad_pedida
  ,pc.precio_ciclo
  ,(ped.cantidad_pedida * pc.precio_ciclo) AS subtotal
	,p.url_imagen

FROM pedidos_detalle ped

JOIN pedidos_cabecera pec
ON ped.pedidos_cabecera_id = pec.id

JOIN productos_ciclo pc 
ON ped.productos_ciclo_id = pc.id

JOIN productos p 
ON pc.producto_id = p.id

WHERE pec.socio_id = {{appsmith.store.socio_id}} 
  AND pec.estado = 'Abierto'
	AND pec.ciclo_id = {{appsmith.store.ciclo_id}}
ORDER BY ped.id
;
