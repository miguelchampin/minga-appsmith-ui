SELECT 
    p.id as producto_id
    ,pc.id as producto_ciclo_id
    ,p.nombre
		,p.descripcion
		,p.clase
		,p.unidad_medida
    ,p.url_imagen
    ,pc.precio_ciclo
    ,p.es_organico
    ,p.es_local
		,p.es_pequena_produccion
		,p.es_mas_barato
		,p.es_vegano
		,p.es_sin_gluten
FROM producto_ciclo pc
JOIN productos p ON pc.producto_id = p.id
JOIN ciclos_compra c ON pc.ciclo_id = c.id
WHERE c.estado = 'Abierto'
--Filtro busqueda
AND p.nombre ILIKE '%' || {{InputBuscador.text || ''}} || '%'
-- Filtro por clase
AND ({{SelectClase.selectedOptionValue === ""}} OR p.clase = {{SelectClase.selectedOptionValue}})
-- Filtros Especiales
AND ({{!FiltrosEspeciales.selectedValues.includes('organico')}} OR p.es_organico = true)
AND ({{!FiltrosEspeciales.selectedValues.includes('local')}} OR p.es_local = true)
AND ({{!FiltrosEspeciales.selectedValues.includes('pequena_produccion')}} OR p.es_pequena_produccion = true)
AND ({{!FiltrosEspeciales.selectedValues.includes('mas_barato')}} OR p.es_mas_barato = true)
AND ({{!FiltrosEspeciales.selectedValues.includes('vegano')}} OR p.es_vegano = true)
AND ({{!FiltrosEspeciales.selectedValues.includes('sin_gluten')}} OR p.es_sin_gluten = true)
;
