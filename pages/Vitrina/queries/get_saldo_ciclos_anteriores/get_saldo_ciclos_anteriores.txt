WITH ciclo_anterior AS (
	SELECT
	id as ciclo_id,
	fecha_apertura
	FROM ciclos_compra
	WHERE estado = 'finalizado'
	ORDER BY fecha_apertura DESC
	LIMIT 1
)
SELECT 
    mcs.id,
    mcs.ciclo_id,
    mcs.tipo_movimiento AS tipo,
    mcs.monto,
    COALESCE(mcs.descripcion, 'Sin detalle') AS descripcion
FROM movimientos_cuenta_socios mcs
INNER JOIN ciclo_anterior c
ON mcs.ciclo_id = c.ciclo_id
WHERE mcs.socio_id = {{appsmith.store.socio_id}}
  AND mcs.estado != 'Anulado'


UNION

SELECT 
    0 as id,
    MAX(mcs.ciclo_id) AS ciclo_id,
    CASE WHEN SUM(CASE WHEN mcs.tipo_movimiento = 'abono' THEN 1 else -1 end * mcs.monto) > 0 THEN 'abono' ELSE 'cargo' END AS tipo,
    ABS(SUM(CASE WHEN mcs.tipo_movimiento = 'abono' THEN 1 else -1 end * mcs.monto)) AS monto,
    'Saldo pendiente' AS descripcion
FROM movimientos_cuenta_socios mcs
INNER JOIN ciclos_compra cc
ON mcs.ciclo_id = cc.id
LEFT JOIN ciclo_anterior ca
ON ca.ciclo_id = cc.id
WHERE ca.ciclo_id IS NULL
	AND cc.estado = 'finalizado'
	AND mcs.socio_id = {{appsmith.store.socio_id}}
  AND mcs.estado != 'Anulado'
	
ORDER BY id ASC
;
