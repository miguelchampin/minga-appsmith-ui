SELECT 
PRD.*
,PRVD.nombre as nombre_proveedor

FROM public."productos" PRD

INNER JOIN public.proveedores PRVD
ON PRD.proveedor_id = PRVD.id

WHERE PRD."nombre" ilike '%{{data_table.searchText || ""}}%'
ORDER BY "{{data_table.sortOrder.column || 'id'}}" {{data_table.sortOrder.order || 'ASC'}}
LIMIT {{data_table.pageSize}}
OFFSET {{(data_table.pageNo - 1) * data_table.pageSize}};
