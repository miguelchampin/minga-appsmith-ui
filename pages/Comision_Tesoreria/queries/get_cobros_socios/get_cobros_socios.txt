SELECT 
    m.id AS movimiento_id,
    s.nombre AS socio,
    n.nombre AS nucleo,
    m.monto,
    m.estado,
    m.comprobante_url
FROM movimientos_cuenta_socios m

JOIN(
	SELECT *
	FROM ciclos_compra
	WHERE fecha_apertura in (select max(fecha_apertura) from ciclos_compra)
)c
on m.ciclo_id = c.id

JOIN socios s 
ON m.socio_id = s.id

LEFT JOIN nucleos n 
ON s.nucleo_id = n.id

WHERE m.tipo_movimiento = 'compra'
ORDER BY n.nombre, s.nombre;
