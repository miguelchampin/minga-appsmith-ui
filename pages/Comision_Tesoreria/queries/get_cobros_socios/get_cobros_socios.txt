SELECT mcs.* /*
    s.id AS socio_id,
    s.nombre,
    SUM(CASE WHEN mcs.tipo_movimiento = 'abono' THEN mcs.monto ELSE 0 END) AS total_abonos,
    SUM(CASE WHEN mcs.tipo_movimiento = 'cargo' THEN mcs.monto ELSE 0 END) AS total_cargos,
    SUM(CASE WHEN mcs.tipo_movimiento = 'abono' THEN mcs.monto ELSE -mcs.monto END) AS saldo_actual
		*/
FROM movimientos_cuenta_socios mcs

INNER JOIN socios s
ON s.id = mcs.socio_id
			
INNER JOIN ciclos_compra cc
ON mcs.ciclo_id = cc.id
	
--GROUP BY s.id, s.nombre
;
